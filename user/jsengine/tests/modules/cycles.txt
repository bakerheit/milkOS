//# case: cycle: hoisted functions work before the exporter runs
//# file: main.js
import { fromB } from './b.js';
log('main', fromB());
//# file: a.js
import { bFn } from './b.js';
log('a runs, b says', bFn());
export function aFn() { return 'A'; }
//# file: b.js
import { aFn } from './a.js';
log('b runs, a says', aFn());
export function bFn() { return 'B'; }
export function fromB() { return 'from b'; }
//# file: boot.js
//# expect
a runs, b says B
b runs, a says A
main from b

//# case: cycle: let binding is in its TDZ until the exporter runs
//# file: main.js
import './a.js';
//# file: a.js
import './b.js';
export let late = 'a-late';
export const alsoLate = 1;
export class K {}
log('a done');
//# file: b.js
import { late, alsoLate, K } from './a.js';
for (const f of [() => late, () => alsoLate, () => K]) {
  try { f(); log('no error'); } catch (e) { log(e.name); }
}
try { typeof late; } catch (e) { log('typeof', e.name); }
log('b done');
//# expect
ReferenceError
ReferenceError
ReferenceError
typeof ReferenceError
b done
a done

//# case: cycle: bindings are live once initialized
//# file: main.js
import { getB, a } from './a.js';
log(getB(), a);
//# file: a.js
import { b } from './b.js';
export let a = 'a1';
export const getB = () => b;
a = 'a2';
//# file: b.js
import { a, getB } from './a.js';
export let b = 'b1';
setTimeoutLike(() => log('later', a, b));
function setTimeoutLike(f) { Promise.resolve().then(f); }
//# expect
b1 a2
later a2 b1

//# case: cycle: namespace of a module that has not run yet
//# file: main.js
import './a.js';
//# file: a.js
import * as b from './b.js';
export let av = 1;
log('a', Object.keys(b).join(','));
//# file: b.js
import * as a from './a.js';
export let bv = 2;
try { log('b', Object.keys(a).join(',')); } catch (e) { log('keys', e.name); }
try { log(a.av); } catch (e) { log(e.name); }
try { Object.keys(a).forEach((k) => a[k]); } catch (e) { log('key', e.name); }
try { log(Object.getOwnPropertyDescriptor(a, 'av')); } catch (e) { log('desc', e.name); }
try { log('av' in a); } catch (e) { log('in', e.name); }
//# expect
keys ReferenceError
ReferenceError
key ReferenceError
desc ReferenceError
true
a bv
