//# case: a module that throws fails its importers
//# file: main.js
import './mid.js';
log('main');
//# file: mid.js
import './bad.js';
log('mid');
//# file: bad.js
log('bad runs');
throw new Error('boom');
//# expect
bad runs
eval error: Error: boom

//# case: a failed module fails the same way every time
//# file: main.js
let first;
for (let i = 0; i < 3; i++) {
  try { await import('./bad.js'); } catch (e) { first ??= e; log(e === first, e.message); }
}
try { await import('./user.js'); } catch (e) { log('via user', e === first); }
//# file: bad.js
log('bad runs once');
throw new Error('boom');
//# file: user.js
import './bad.js';
//# expect
bad runs once
true boom
true boom
true boom
via user true

//# case: a thrown non-error value is kept as is
//# file: main.js
try { await import('./bad.js'); } catch (e) { log(e === 42, typeof e); }
//# file: bad.js
throw 42;
//# expect
true number

//# case: a module that threw halfway leaves its earlier exports initialized
//# file: main.js
import * as ns from './bad.js';
//# file: bad.js
export let a = 'set';
export function f() { return 1; }
throw new Error('x');
//# expect
eval error: Error: x

//# case: a failed dependency does not run later siblings
//# file: main.js
import './ok1.js';
import './bad.js';
import './ok2.js';
//# file: ok1.js
log('ok1');
//# file: bad.js
throw new Error('bad');
//# file: ok2.js
log('ok2');
//# expect
ok1
eval error: Error: bad

//# case: a missing export is a link error that names the import
//# file: main.js
log('never runs');
import { nope } from './m.js';
//# file: m.js
export const yes = 1;
//# expect
link error: SyntaxError: The requested module './m.js' does not provide an export named 'nope'

//# case: a missing default export
//# file: main.js
import d from './m.js';
//# file: m.js
export const yes = 1;
//# expect
link error: SyntaxError: The requested module './m.js' does not provide an export named 'default'

//# case: a missing re-export is a link error in the re-exporting module
//# file: main.js
import './mid.js';
//# file: mid.js
export { nope } from './m.js';
//# file: m.js
export const yes = 1;
//# expect
link error: SyntaxError: The requested module './m.js' does not provide an export named 'nope'

//# case: export star of a missing name is only an error when imported
//# file: main.js
import * as ns from './mid.js';
log(Object.keys(ns).join());
//# file: mid.js
export * from './m.js';
//# file: m.js
export const yes = 1;
//# expect
yes

//# case: a link error leaves nothing half-linked
//# file: main.js
log('start');
const tries = [];
for (let i = 0; i < 2; i++) {
  try { await import('./bad-link.js'); } catch (e) { tries.push(e.name); }
}
log(tries.join());
try { const m = await import('./good.js'); log(m.v); } catch (e) { log('good failed ' + e.name); }
//# file: bad-link.js
import { missing } from './dep.js';
log('never');
//# file: good.js
import { v } from './dep.js';
export { v };
//# file: dep.js
export const v = 'good value';
//# expect
start
SyntaxError,SyntaxError
good value

//# case: duplicate export names
//# file: main.js
import './x.js';
//# file: x.js
export const a = 1;
export { a };
//# expect
load error: SyntaxError: Duplicate export of 'a'

//# case: duplicate export default
//# file: main.js
import './x.js';
//# file: x.js
export default 1;
export default 2;
//# expect
load error: SyntaxError: Duplicate export of 'default'

//# case: duplicate import binding
//# file: main.js
import './x.js';
//# file: x.js
import { a } from './y.js';
import { b as a } from './y.js';
//# expect
load error: SyntaxError: Identifier 'a' has already been declared

//# case: import conflicts with a declaration
//# file: main.js
import './x.js';
//# file: x.js
import { a } from './y.js';
let a = 1;
//# expect
load error: SyntaxError: Identifier 'a' has already been declared

//# case: function and var clash at module level
//# file: main.js
import './x.js';
//# file: x.js
var f = 1;
function f() {}
//# expect
load error: SyntaxError: Identifier 'f' has already been declared

//# case: var may repeat at module level
//# file: main.js
var a = 1;
var a = 2;
log(a);
//# expect
2

//# case: import and export only at the top level
//# file: main.js
import './x.js';
//# file: x.js
{
  import { a } from './y.js';
}
//# expect
load error: SyntaxError: Cannot use import statement here: imports are only allowed at the top level of a module
